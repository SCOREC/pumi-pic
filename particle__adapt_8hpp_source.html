120template<int dim>
121OMEGA_H_DEVICE double dist2Plane(OH::Matrix<dim, 3> tri, OH::Vector<dim> pos) {
122 if constexpr (dim == 3) {
123 auto normal = OH::get_triangle_normal(tri[0], tri[1], tri[2]);
124 auto dist = (pos - tri[0]) * (normal) / OH::norm(normal);
125 return dist;
126 } else return 0;
127}

129template<int dim>
130int compareWithPosition(PADAPT<dim>& pAdapt) {
131 pAdapt.update(pAdapt.mesh);
132 auto vert2coords = pAdapt.mesh.coords();
133 auto edge2verts = pAdapt.mesh.ask_verts_of(OH::EDGE);
134 auto face2verts = pAdapt.mesh.ask_verts_of(OH::FACE);
135 PS::kkLidView failed = PS::kkLidView("failed", 1);
136 auto getNewElement = PS_LAMBDA(const int& e, const int& pid, const int& mask) {
137 if (mask <= 0) return;
138 auto pPos = pAdapt.getPos(pid);
139 if (pAdapt.pDim(pid) == OH::VERT) {
140 auto verts = OH::gather_verts<dim+1>(pAdapt.downward[OH::VERT].ab2b, OH::LO(pAdapt.pParent(pid)));
141 auto coords = OH::gather_vectors<dim+1,dim>(vert2coords, verts);
142 if (!OH::are_close(coords[pAdapt.pChild(pid)], pPos)) {
143 printf("[ERROR] Particle %d not at correct vertex\n", pid);
144 failed(0) = 1;
145 }
146 }
147 else if (pAdapt.pDim(pid) == OH::EDGE) {
148 auto child = pAdapt.getChildElem(pid);
149 auto eVerts = OH::gather_verts<2>(edge2verts, child);
150 auto eCoords = OH::gather_vectors<2, dim>(vert2coords, eVerts);
151 auto baryCoords = OH::barycentric_from_global<dim,1>(pPos, eCoords);
152 if (!OH::is_barycentric_inside(baryCoords, OH::EPSILON)){
153 printf("[ERROR] Particle %d is on edge %d which is not correct\n", pid, child);
154 failed(0) = 1;
155 }
156 }
157 else if (pAdapt.pDim(pid) == OH::FACE) {
158 auto child = pAdapt.getChildElem(pid);
159 auto eVerts = OH::gather_verts<3>(face2verts, child);
160 auto eCoords = OH::gather_vectors<3, dim>(vert2coords, eVerts);
161 auto baryCoords = OH::barycentric_from_global<dim,2>(pPos, eCoords);
162 auto dist = dist2Plane(eCoords, pPos);
163 if (!OH::are_close(dist, 0) || !OH::is_barycentric_inside(baryCoords, OH::EPSILON)){
164 printf("[ERROR] Particle %d is on face %d which is not correct\n", pid, child);
165 failed(0) = 1;
166 }
167 }
168 auto parentVerts = OH::gather_verts<dim+1>(pAdapt.downward->ab2b, OH::LO(pAdapt.pParent(pid)));
169 auto parentCoords = OH::gather_vectors<dim+1,dim>(vert2coords, parentVerts);
170 auto baryCoords = OH::barycentric_from_global<dim,dim>(pPos, parentCoords);
171 if (OH::is_barycentric_inside(baryCoords, OH::EPSILON)) return;
172 printf("[ERROR] Particle %d is not in parent %d\n", pid, pAdapt.pParent(pid));
173 failed(0) = 1;
174 };
175 ps::parallel_for(pAdapt.ptcls, getNewElement);
176 return ps::getLastValue(failed);
177}

210int isParticleInLowest(PADAPT<dim>& pAdapt) {
211 pAdapt.update(pAdapt.mesh);
212 PS*& ptcls = pAdapt.ptcls;
213 auto ptclID = ptcls->get<PID>();
214 PS::kkLidView failed = PS::kkLidView("failed", 1);
215 auto printResults = PS_LAMBDA(const int& e, const int& pid, const int& mask) {
216 if (mask <= 0 || pAdapt.pDim(pid) == dim) return;
217 auto child = pAdapt.getChildElem(pid);
218 auto lowestElem = pAdapt.getLowestParent(child, pAdapt.pDim(pid));
219 if (pAdapt.pParent(pid) == lowestElem) return;
220 printf("Ptcl %-2d: Not on lowest parent. Is (%-2d) should be (%-2d)\n", ptclID(pid), pAdapt.pParent(pid), lowestElem);
221 failed(0) = 1;
222 };
223 ps::parallel_for(pAdapt.ptcls, printResults);
224 return ps::getLastValue(failed);

241 auto test_ent2verts = pAdapt.mesh.get_adj(test_dim, OH::VERT).ab2b;
