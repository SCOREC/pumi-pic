174 // Returns a copy of in_baryCoords that has been modified the boundary of an entity
175 template <int n>
176 OMEGA_H_DEVICE Vector<n> move2Entity(const Vector<n> in_baryCoords, Int dim, Int entIdx) const {
177 Vector<n> baryCoords = in_baryCoords;
178 bool on_ent[mesh_dim + 1] = {false};
179 for (Int v = 0; v < simplex_degree(dim, VERT); ++v)
180 on_ent[simplex_down_template(mesh_dim, dim, entIdx, v)] = true;
181
182 Real barySum = 0;
183 for (Int i = 0; i < mesh_dim + 1; ++i)
184 if (!on_ent[i] || baryCoords[i] < 0) baryCoords[i] = 0;
185 else barySum += baryCoords[i];
186
187 if (are_close(barySum, 0))
188 for (Int i = 0; i < mesh_dim + 1; ++i)
189 if (on_ent[i]) baryCoords[i] = 1;
190
191 return pp::clamp_barycentric<mesh_dim>(baryCoords);
192 }

194 // Modifies baryCoords to be on the boundary of the closest adjacent element classified at target class id
195 template <int n>
196 OMEGA_H_DEVICE bool move2ClosestModelTarget(Vector<n> &baryCoords, LO elem, I8 old_class_dim, ClassId target) const {
197 Int dimClosest = 0;
198 Int idxClosest = 0;
199 Real closest = ArithTraits<Real>::max();
200
201 for (Int dim = mesh_dim - 1; dim >= 0; --dim) {
202 auto nEnts = simplex_degree(mesh_dim, dim);
203 for (Int entIdx = 0; entIdx < nEnts; ++entIdx) {
204 auto entID = downward[dim].ab2b[elem * nEnts + entIdx];
205 if (class_id[dim][entID] != target) continue;
206 auto newBaryCoords = move2Entity(baryCoords, dim, entIdx);
207 auto dist = norm(newBaryCoords - baryCoords);
208 if (dist >= closest) continue;
209 closest = dist;
210 dimClosest = dim;
211 idxClosest = entIdx;
212 }
213 }
214
215 if (closest == ArithTraits<Real>::max()) return false;
216 baryCoords = move2Entity(baryCoords, dimClosest, idxClosest);
217 return true;
218 }

220 // Returns distance to closest adjacent element classified at target class id
221 OMEGA_H_DEVICE Real barycentric_distance(const LO pid, const LO elem, const I8 old_class_dim, const ClassId target) const {
222 auto verts = gather_verts<mesh_dim+1>(downward[VERT].ab2b, elem);
223 auto coords = gather_vectors<mesh_dim+1,mesh_dim>(vert2coords, verts);
224 auto baryCoords = barycentric_from_global<mesh_dim,mesh_dim>(getPos(pid), coords);
225
226 if (old_class_dim < mesh_dim && !move2ClosestModelTarget(baryCoords, elem, old_class_dim, target))
227 return ArithTraits<Real>::max();
228 else baryCoords = pp::clamp_barycentric<mesh_dim>(baryCoords);
229
230 OMEGA_H_CHECK(is_barycentric_inside(baryCoords, EPSILON));
231 auto newPosition = pp::global_from_barycentric<mesh_dim,mesh_dim>(baryCoords, coords);
232 return norm(newPosition - getPos(pid));
233 }

235 OMEGA_H_DEVICE void assign2Elem(const LO pid, const LO elem) const {
236 auto verts = gather_verts<mesh_dim+1>(downward[VERT].ab2b, elem);
237 auto coords = gather_vectors<mesh_dim+1,mesh_dim>(vert2coords, verts);
238 auto baryCoords = barycentric_from_global<mesh_dim,mesh_dim>(getPos(pid), coords);
239 OMEGA_H_CHECK(is_barycentric_inside(baryCoords, EPSILON));
240 pParent(pid) = elem;
241 pDim(pid) = mesh_dim;
242
243 for (Int dim = 0; dim < mesh_dim; dim++)
244 for (Int ent = 0; ent < simplex_degree(mesh_dim, dim); ent++) {
245 Real baryCoordsSum = 0.0;
246 for (Int vert = 0; vert < simplex_degree(dim, VERT); vert++) {
247 auto vertIdx = simplex_down_template(mesh_dim, dim, ent, vert);
248 if (are_close(baryCoords[vertIdx], 0)) {baryCoordsSum = -100.0; break;}
249 else baryCoordsSum += baryCoords[vertIdx];
250 }
251 if (!are_close(baryCoordsSum, 1.0)) continue;
252 pDim(pid) = dim;
253 pChild(pid) = ent;
254 }
255 update2LowestParent(pid);
256 }

279 virtual void refine(Mesh& old_mesh, Mesh& new_mesh, LOs keys2edges, LOs keys2midverts, Int prod_dim,
280 LOs keys2prods, LOs prods2new_ents, LOs same_ents2old_ents, LOs same_ents2new_ents) {
281 if (prod_dim != mesh_dim) return;
282 auto old2New = getUnchanged(old_mesh, prod_dim, same_ents2old_ents, same_ents2new_ents);
283 auto modified = gatherModified(old_mesh, keys2edges, EDGE);
284 auto old_cell2verts = old_mesh.ask_down(mesh_dim, VERT).ab2b;
285 auto old_vert2coords = old_mesh.coords();
286 update(new_mesh);
287
288 //Update modified elements
289 Kokkos::parallel_for(ptcls->nPtcls(), KOKKOS_CLASS_LAMBDA(const LO pid) {
290 auto oldElem = pParent(pid);
291 if (old2New[oldElem] != -1) {//update unchanged element id
292 pParent(pid) = old2New[oldElem];
293 update2LowestParent(pid);
294 }
295 else if (modified[oldElem].offset != -1) { //find new split element
296 auto newVert = mesh_dim;
297 auto rotation = code_rotation(modified[oldElem].code);
298 auto spltEdgeIdx = code_which_down(modified[oldElem].code);
299 auto spltVerts = ps::simplex_gather_down<EDGE>(mesh_dim, spltEdgeIdx, rotation);
300 auto oldVerts = gather_verts<mesh_dim+1>(old_cell2verts, oldElem);
301 auto oldCoords = gather_vectors<mesh_dim+1,mesh_dim>(old_vert2coords, oldVerts);
302 auto baryCoords = barycentric_from_global<mesh_dim,mesh_dim>(getPos(pid), oldCoords);
303 bool onSplit = are_close(baryCoords[spltVerts[0]], baryCoords[spltVerts[1]]) && !are_close(baryCoords[spltVerts[0]], 0);
304 auto target = (onSplit || baryCoords[spltVerts[1]] > baryCoords[spltVerts[0]]) ? 0 : 1;
305 auto prod = keys2prods[modified[oldElem].key] + modified[oldElem].offset*2 + target;
306 pParent(pid) = prods2new_ents[prod];
307
308 auto keptSide = simplex_opposite_template(mesh_dim, VERT, spltVerts[target]);
309 Int old2NewIdx[mesh_dim+1] = {0}; //one elem kept blank
310 for (Int newIdx = 0; newIdx < mesh_dim; ++newIdx) {
311 auto oldIdx = simplex_down_template(mesh_dim, mesh_dim - 1, keptSide, newIdx);
312 old2NewIdx[oldIdx] = flip_new_vert(mesh_dim, newIdx);
313 }
314
315 if (onSplit) pDim(pid) = pDim(pid) - 1;
316 if (onSplit && pDim(pid) == VERT) { //ptcl on the new vert
317 pChild(pid) = newVert;
318 }
319 else if (onSplit && pDim(pid) == EDGE) { //ptcl on a new edge
320 auto oppositeVert = pp::face_vertex_opposite_edge(mesh_dim, pChild(pid), spltEdgeIdx);
321 pChild(pid) = pp::edge_from_verts(mesh_dim, newVert, old2NewIdx[oppositeVert]);
322 }
323 else if (onSplit && pDim(pid) == FACE) { //particle on a new face
324 auto oppositeEdge = simplex_opposite_template(mesh_dim, EDGE, spltEdgeIdx);
325 auto oppositeVerts = pp::simplex_gather_down<EDGE>(mesh_dim, oppositeEdge);
326 auto edge1 = pp::edge_from_verts(mesh_dim, newVert, old2NewIdx[oppositeVerts[0]]);
327 auto edge2 = pp::edge_from_verts(mesh_dim, newVert, old2NewIdx[oppositeVerts[1]]);
328 pChild(pid) = pp::face_from_edges(edge1, edge2);
329 }
330 else if (pDim(pid) == VERT) { //ptcl stayed on same vert
331 pChild(pid) = old2NewIdx[pChild(pid)];
332 }
333 else if (pDim(pid) == EDGE) { //ptcl stayed on same edge
334 auto edgeVerts = pp::simplex_gather_down<EDGE>(mesh_dim, pChild(pid));
335 pChild(pid) = (pChild(pid) == spltEdgeIdx) ?
336 pp::edge_from_verts(mesh_dim, newVert, old2NewIdx[spltVerts[1-target]]) : //old edge was split
337 pp::edge_from_verts(mesh_dim, old2NewIdx[edgeVerts[0]], old2NewIdx[edgeVerts[1]]); //old edge stayed the same
338 }
339 else if (pDim(pid) == FACE && pDim(pid) < mesh_dim) { //particle stayed on face
340 if (are_close(baryCoords[spltVerts[0]], 0) || are_close(baryCoords[spltVerts[1]], 0)) { //old face stayed the same
341 auto faceVerts = pp::simplex_gather_down<FACE>(mesh_dim, pChild(pid));
342 auto edge1 = pp::edge_from_verts(mesh_dim, old2NewIdx[faceVerts[0]], old2NewIdx[faceVerts[1]]);
343 auto edge2 = pp::edge_from_verts(mesh_dim, old2NewIdx[faceVerts[0]], old2NewIdx[faceVerts[2]]);
344 pChild(pid) = pp::face_from_edges(edge1, edge2);
345 }
346 else { //old face was split
347 auto oppositeVert = pp::face_vertex_opposite_edge(mesh_dim, pChild(pid), spltEdgeIdx);
348 auto edge1 = pp::edge_from_verts(mesh_dim, newVert, old2NewIdx[spltVerts[1-target]]);
349 auto edge2 = pp::edge_from_verts(mesh_dim, newVert, old2NewIdx[oppositeVert]);
350 pChild(pid) = pp::face_from_edges(edge1, edge2);
351 }
352 }
353 update2LowestParent(pid);
354 }
355 else Kokkos::abort("[ERROR] : element skipped during particle adaptation\n");
356 });

359 void updatePtclsCavitySearch(Mesh& old_mesh, Mesh& new_mesh, LOs keys2prods, LOs prods2new_ents,
360 LOs same_ents2old_ents, LOs same_ents2new_ents, Kokkos::View<ModifiedElem*> modified_elem) {
361 update(new_mesh);
362 auto old2New = getUnchanged(old_mesh, mesh_dim, same_ents2old_ents, same_ents2new_ents);
363 MeshData old_data = MeshData<mesh_dim>(old_mesh);
364 Kokkos::parallel_for(ptcls->nPtcls(), KOKKOS_CLASS_LAMBDA(const int pid) {
365 auto oldElem = pParent(pid);
366 auto newElem = oldElem;
367 auto oldChild = getChildElem(pid, old_data.downward);
368 auto oldClassDim = old_data.class_dim[pDim(pid)][oldChild];
369 auto oldClassID = old_data.class_id[pDim(pid)][oldChild];
370
371 if (old2New[oldElem] != -1)
372 newElem = old2New[oldElem];
373 else if (modified_elem[oldElem].key != -1) {
374 Real closest = ArithTraits<Real>::max();
375 auto key = modified_elem[oldElem].key;
376 for (auto idx = keys2prods[key]; idx < keys2prods[key+1]; ++idx) {
377 auto dist = barycentric_distance(pid, prods2new_ents[idx], oldClassDim, oldClassID);
378 if (dist < closest) {closest = dist; newElem = prods2new_ents[idx];}
379 }
380 OMEGA_H_CHECK(closest != ArithTraits<Real>::max());
381 }
382 else Kokkos::abort("[ERROR] : particle skipped during particle adaptation of swap/coarsen\n");
383
384 snap2Surface(pid, newElem, oldClassDim, oldClassID);
385 assign2Elem(pid, newElem);
386 });

389 virtual void snap(Mesh& mesh, const Omega_h::Reals& old_vert2coords, const Omega_h::Reals& warp) {
390 update(mesh);
391 auto adjElems = mesh.ask_dual();
392 Kokkos::parallel_for(ptcls->nPtcls(), KOKKOS_CLASS_LAMBDA(const int pid) {
393 auto elem = pParent(pid);
394 auto child = getChildElem(pid);
395 auto classID = class_id[pDim(pid)][child];
396 auto classDim = class_dim[pDim(pid)][child];
397 Real closest = barycentric_distance(pid, elem, classDim, classID);
398 LO closestElem = elem;
399 for (int adj = adjElems.a2ab[elem]; adj < adjElems.a2ab[elem+1]; adj++) {
400 auto dist = barycentric_distance(pid, adjElems.ab2b[adj], classDim, classID);
401 if (dist >= closest) continue;
402 dist = closest;
403 closestElem = adjElems.ab2b[adj];
404 }
405 snap2Surface(pid, closestElem, classDim, classID);
406 assign2Elem(pid, closestElem);
407 });
408 }
